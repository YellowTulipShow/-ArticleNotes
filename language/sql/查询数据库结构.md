# 查询数据库所有表名

```sql

select * from (
  SELECT
      TableName = CASE WHEN C.column_id=1 THEN O.name ELSE N'' END,
      TableExplain = CONVERT(nvarchar(MAX), ISNULL(CASE WHEN C.column_id=1 THEN PTB.[value] END,N''))
  FROM sys.columns C
  INNER JOIN sys.objects O ON C.[object_id]=O.[object_id] AND C.column_id=1 AND O.type='U' AND O.is_ms_shipped=0
  LEFT JOIN sys.extended_properties PTB ON PTB.class=1 AND PTB.minor_id=0 AND C.[object_id]=PTB.major_id
) as T ORDER BY T.TableName asc

```

# 查询数据库下指定表名的结构

```sql

-- 临时过程: 查询目标下菜单
IF OBJECT_ID('tempdb..#QueryTableContents') IS NOT NULL
BEGIN
drop PROCEDURE #QueryTableContents
END
IF OBJECT_ID('tempdb..#QueryTableContents') IS NULL
BEGIN
EXEC('
    CREATE PROCEDURE #QueryTableContents @TableName NVARCHAR(MAX) AS
    BEGIN
        SELECT
            @TableName AS 字段名称,
            (select top 1 T.value from sys.extended_properties as T where major_id = OBJECT_ID(@TableName) and minor_id = 0) AS 说明,
            NULL AS 数据类型,
            NULL AS 是否空值,
            NULL AS 长度
        FROM (VALUES (1)) AS T(Col)
        UNION ALL
        select
            field.name as 字段名称,
            extendedProperties.value as 说明,
            (select top 1 name from sys.types where field.system_type_id = system_type_id) as 数据类型,
            field.is_nullable as 是否空值,
            field.max_length as 长度
        from
            sys.columns as field
            left join sys.extended_properties as extendedProperties
                on(major_id = OBJECT_ID(@TableName) and field.column_id = extendedProperties.minor_id)
        where object_id = (select top 1 object_id from sys.tables where name = @TableName)
    END
');
END

-- 注意: 不要加数据库 USE *** 引用
```
